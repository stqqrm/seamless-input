#include "input/input.hpp"
#include "input/input_backend.hpp"
#include <cassert>
#include <chrono>
#include <cstdio>
#include <thread>
#include <atomic>

struct Fake : InputBackend {
    const char* Name() const override { return "fake"; }
    bool Start(BackendSink s, std::string*) override { sink = std::move(s); return true; }
    void Stop() override {}
    bool BeginForwarding(int px, int py, std::string*) override { begun++; parkX=px; parkY=py; return true; }
    void EndForwarding(int x, int y) override { ended++; endX=x; endY=y; }
    void InjectMoveTo(int x, int y) override { cx=x; cy=y; }
    void InjectButton(int b, bool d) override { btn=b*10+d; }
    void InjectWheel(int, int) override {}
    void InjectKey(int c, bool d) override { lastKey=c; keyDown=d; }
    BackendSink sink; std::atomic<int> begun{0}, ended{0}; int parkX=0,parkY=0,endX=-1,endY=-1,cx=-1,cy=-1,btn=-1,lastKey=-1; bool keyDown=false;
};
static Fake* g_fake[2]; static int g_n = 0;
std::unique_ptr<InputBackend> MakeEvdevBackend() { auto f = std::make_unique<Fake>(); g_fake[g_n++] = f.get(); return f; }
std::unique_ptr<InputBackend> MakeX11Backend(std::string*) { return nullptr; }

#define CHECK(c) do { if(!(c)) { std::printf("FAIL line %d: %s\n", __LINE__, #c); return 1; } } while(0)
static void settle() { std::this_thread::sleep_for(std::chrono::milliseconds(60)); }

int main() {
    setenv("SEAMLESS_INPUT_BACKEND", "evdev", 1);
    InputEngine *A = nullptr, *B = nullptr;
    InputEngine a([&](int, const std::string& l){ B->HandleMessage(1, l); }, [](const std::string& s){ std::printf("A: %s\n", s.c_str()); });
    InputEngine b([&](int, const std::string& l){ A->HandleMessage(1, l); }, [](const std::string& s){ std::printf("B: %s\n", s.c_str()); });
    A = &a; B = &b;
    std::vector<InputRect> am{{0,0,1920,1080}}, bm{{0,0,1280,720}};
    a.SetLocalMonitors(am); b.SetLocalMonitors(bm);
    std::string m; CHECK(a.Start(&m)); CHECK(b.Start(&m));
    Fake &fa = *g_fake[0], &fb = *g_fake[1];
    a.SetPeer(1, bm, 1920, 0);  b.SetPeer(1, am, -1920, 0);

    // 1. B does not allow A and A has not been told otherwise: touching the edge must do NOTHING
    //    (no grab, no cursor parked in the middle of the screen, nothing sent).
    fa.sink.onLocalMotion(1919, 500, false); settle();
    CHECK(fa.begun == 0); CHECK(fa.ended == 0);

    // 1b. stale permission (A believes it may, B refuses): ENTER -> DENY -> A gets its input back once,
    //     and A does not try again until B says CTRL 1.
    a.SetPeerAccepts(1, true);
    fa.sink.onLocalMotion(1919, 500, false); settle();
    CHECK(fa.begun == 1); CHECK(fa.ended == 1);
    std::this_thread::sleep_for(std::chrono::milliseconds(300));   // cooldown
    fa.sink.onLocalMotion(1919, 500, false); settle();
    CHECK(fa.begun == 1);

    // 2. allow (B tells A with CTRL 1), then cross for real
    b.SetPeerAllowed(1, true);
    a.SetPeerAccepts(1, true);
    fa.sink.onLocalMotion(1000, 500, false);       // middle of the screen: nothing
    CHECK(fa.begun == 1);
    fa.sink.onLocalMotion(1919, 500, false); settle();
    CHECK(fa.begun == 2);
    CHECK(fb.cx == 4 && fb.cy == 500);             // entered B at its left edge (+inset), same y
    std::printf("park=(%d,%d)\n", fa.parkX, fa.parkY); CHECK(fa.parkX == 960 && fa.parkY == 540);

    // 3. forwarded events arrive on B
    fa.sink.onForwarded({InputKind::Move, 100, -20}); settle();
    CHECK(fb.cx == 104 && fb.cy == 480);
    fa.sink.onForwarded({InputKind::Button, 0, 1}); settle(); CHECK(fb.btn == 1);
    fa.sink.onForwarded({InputKind::Key, 30, 1}); settle(); CHECK(fb.lastKey == 30 && fb.keyDown);

    // 4. push left through B's left edge -> control returns to A at its right edge (inset)
    fa.sink.onForwarded({InputKind::Move, -400, 0}); settle();
    std::printf("A end=(%d,%d) ended=%d\n", fa.endX, fa.endY, fa.ended.load());
    CHECK(fa.ended == 2);
    CHECK(fa.endX == 1919 - 4 && fa.endY == 480);
    CHECK(fb.lastKey == 30 && !fb.keyDown);        // held key released on B when control left

    // 5. Scroll Lock emergency exit
    std::this_thread::sleep_for(std::chrono::milliseconds(300));
    fa.sink.onLocalMotion(1919, 300, false); settle(); CHECK(fa.begun == 3);
    fa.sink.onForwarded({InputKind::Key, 70, 1}); settle();
    CHECK(fa.ended == 3); CHECK(fa.endX == 1919 - 24);

    // 6. vertical gap: B placed below-right, only the overlapping y-range crosses
    a.SetPeer(1, bm, 1920, 600);  b.SetPeer(1, am, -1920, -600);
    std::this_thread::sleep_for(std::chrono::milliseconds(300));
    fa.sink.onLocalMotion(1919, 100, false); settle(); CHECK(fa.begun == 3);   // y=100 is above B
    fa.sink.onLocalMotion(1919, 700, false); settle(); CHECK(fa.begun == 4);
    CHECK(fb.cx == 4 && fb.cy == 100);             // world y 700 -> B y 100

    a.Stop(); b.Stop();
    std::puts("ALL OK");
}
